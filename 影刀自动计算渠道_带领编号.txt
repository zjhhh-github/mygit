def 是合法编号(编号):
    """
    判断传入内容是否为合法编号。

    合法编号规则：
    1. 不能为空
    2. 去掉头尾空格后，必须是连续 6 位数字
    """

    if 编号 is None:
        return False

    编号 = str(编号).strip()

    return re.fullmatch(r"\d{6}", 编号) is not None

# ============================================================
# 五、核心算法：查找推荐人编号和渠道B编号
# ============================================================
# 调试开关：排查"推荐人有值、渠道B为空"等异常时打开；正式运行可置为 False
调试日志_查找推荐人和渠道B = True

def 查找推荐人和渠道B(报名学员编号):
    """
    根据一个报名学员编号，向上查找它最终对应的：
    1. 推荐人编号
    2. 渠道B编号
    3. 特殊指定带领B编号（仅在向上查找过程中命中"特殊渠道带领指定表"时返回；否则为 None）

    说明：
    - 推荐人编号：只取第一次查到的上一级推荐人编号（不受特殊表影响）
    - 渠道B编号：
        * 若向上查找过程中任意一层命中特殊表，则渠道B编号 = 该层特殊表 value[0]，并立即结束查找
        * 否则按原规则向上找到符合条件的合伙宝妈编号
    - 特殊指定带领B编号：
        * 命中特殊表时 = 该层特殊表 value[1]（即使是空串也原样返回）
        * 未命中时 = None；调用方据此决定带领B 是走特殊指定值还是走 计算带领B编号(...)

    特殊表命中时机：
    - 在循环每一层的开头，先用"当前报名学员编号"去查特殊表
    - 这样原始报名学员编号、链路上的中间节点（含合伙宝妈）都有机会命中
    - 谁先命中，就以谁的指定值为准
    """

    推荐人编号 = ""
    渠道B编号 = ""

    if 调试日志_查找推荐人和渠道B:
        print("[查找渠道B-入口] 报名学员编号={}".format(报名学员编号))

    if not 是合法编号(报名学员编号):
        if 调试日志_查找推荐人和渠道B:
            print("[查找渠道B-返回空] 入口编号不合法，报名学员编号={}".format(报名学员编号))
        return 推荐人编号, 渠道B编号, None

    原始报名学员编号 = str(报名学员编号).strip()
    当前报名学员编号 = 原始报名学员编号

    已访问编号集合 = set()
    _轮次 = 0

    while True:
        _轮次 += 1
        # 每一层先判断特殊表命中（原始报名学员编号 -> 第一次推荐人 -> 中间合伙宝妈 ...）
        # 命中后立即结束渠道B / 带领B 的向上查找
        指定渠道B编号, 指定带领B编号 = 获取特殊指定渠道B带领B(当前报名学员编号)
        if 指定渠道B编号 is not None:
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-特殊表命中] 第{}轮 当前报名学员编号={}，指定渠道B={}，指定带领B={}".format(
                        _轮次, 当前报名学员编号, 指定渠道B编号, 指定带领B编号
                    )
                )
            return 推荐人编号, 指定渠道B编号, 指定带领B编号

        if 当前报名学员编号 in 已访问编号集合:
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-返回空] 第{}轮 当前编号已访问过（防环），当前报名学员编号={}，已确定推荐人编号={}".format(
                        _轮次, 当前报名学员编号, 推荐人编号
                    )
                )
            return 推荐人编号, "", None

        已访问编号集合.add(当前报名学员编号)

        if not 是合法编号(当前报名学员编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-返回空] 第{}轮 当前编号不合法，当前报名学员编号={}，已确定推荐人编号={}".format(
                        _轮次, 当前报名学员编号, 推荐人编号
                    )
                )
            return 推荐人编号, "", None

        当前报名学员编号 = str(当前报名学员编号).strip()

        上一级推荐人编号 = 学员来源映射.get(当前报名学员编号, "")

        if not 是合法编号(上一级推荐人编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-返回空] 第{}轮 上一级推荐人不合法，当前报名学员编号={}，上一级推荐人编号={}，已确定推荐人编号={}".format(
                        _轮次, 当前报名学员编号, 上一级推荐人编号, 推荐人编号
                    )
                )
            return 推荐人编号, "", None

        上一级推荐人编号 = str(上一级推荐人编号).strip()

        # 推荐人编号只保留第一次查到的结果
        # 后续继续向上查找，只用于确定渠道B编号
        _首次写入推荐人 = (推荐人编号 == "")
        if _首次写入推荐人:
            推荐人编号 = 上一级推荐人编号

        if 调试日志_查找推荐人和渠道B:
            print(
                "[查找渠道B-轮次] 第{}轮 当前报名学员编号={}，上一级推荐人编号={}，是否首次写入推荐人={}".format(
                    _轮次, 当前报名学员编号, 上一级推荐人编号, _首次写入推荐人
                )
            )

        前五学员编号列表 = 宝妈前五学员映射.get(上一级推荐人编号)

        # 当前推荐人不是合伙宝妈，继续向上找
        if 前五学员编号列表 is None:
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-非合伙宝妈] 第{}轮 上一级推荐人={} 不在 宝妈前五学员映射，继续向上找".format(
                        _轮次, 上一级推荐人编号
                    )
                )
            当前报名学员编号 = 上一级推荐人编号
            continue

        合法前五学员编号集合 = set()

        for 前五学员编号 in 前五学员编号列表:
            if 是合法编号(前五学员编号):
                合法前五学员编号集合.add(str(前五学员编号).strip())

        _当前是否在前五 = 当前报名学员编号 in 合法前五学员编号集合
        if 调试日志_查找推荐人和渠道B:
            print(
                "[查找渠道B-命中前五] 第{}轮 命中合伙宝妈={}，前五学员={}，当前报名学员编号={}，是否在前五={}".format(
                    _轮次, 上一级推荐人编号, 前五学员编号列表, 当前报名学员编号, _当前是否在前五
                )
            )

        # 当前节点编号不在前五，则当前合伙宝妈就是渠道B
        if not _当前是否在前五:
            渠道B编号 = 上一级推荐人编号
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-命中] 第{}轮 当前节点不在前五，渠道B={}".format(_轮次, 渠道B编号)
                )
            return 推荐人编号, 渠道B编号, None

        # 方案A 直属放行（精准收窄版）：必须同时满足两个条件才放行：
        # 1) 本轮"当前报名学员编号"就是入口"原始报名学员编号"（即未经过任何向上跳跃）
        # 2) 上一级推荐人编号 是"本轮运行中"通过 初始扫描兜底 / 动态晋升 补录出来的合伙宝妈
        # 这样：飞书原本就有的老合伙宝妈完全不会被该规则影响（保持原算法），
        # 仅修复"新增学员触发动态晋升后被塞进直属推荐人前五，导致渠道B为空"这一精确场景。
        if (
            当前报名学员编号 == 原始报名学员编号
            and 上一级推荐人编号 in 本轮新增合伙宝妈编号集合
        ):
            渠道B编号 = 上一级推荐人编号
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[查找渠道B-直属放行] 第{}轮 原始报名学员首次命中【本轮新增】直属合伙宝妈，虽然在前五中，仍直接认定渠道B={}".format(
                        _轮次, 渠道B编号
                    )
                )
            return 推荐人编号, 渠道B编号, None

        # 当前节点编号在前五，继续向上找
        # 注：下一轮循环开头会用新的"当前报名学员编号"（即此处的合伙宝妈编号）再去判断特殊表
        if 调试日志_查找推荐人和渠道B:
            print(
                "[查找渠道B-继续向上] 第{}轮 当前节点在前五，继续向上找，上移到={}".format(
                    _轮次, 上一级推荐人编号
                )
            )
        当前报名学员编号 = 上一级推荐人编号


# ============================================================
# 六、计算带领B编号
# ============================================================

def 计算带领B编号(报名学员编号, 渠道B编号):
    """
    根据报名学员编号和渠道B编号，计算最终带领B编号。

    规则：
    1. 先用渠道B编号去通用带领B指定映射表中匹配
    2. 如果没匹配到，再用报名学员编号去个性带领B指定映射表中匹配
    3. 如果都没匹配到，则带领B编号等于渠道B编号
    """

    if not 是合法编号(渠道B编号):
        return ""

    渠道B编号 = str(渠道B编号).strip()

    通用指定带领B编号 = 通用带领B指定映射表.get(渠道B编号, "")

    if 是合法编号(通用指定带领B编号):
        return str(通用指定带领B编号).strip()

    if 是合法编号(报名学员编号):
        报名学员编号 = str(报名学员编号).strip()

        个性指定带领B编号 = 个性带领B指定映射表.get(报名学员编号, "")

        if 是合法编号(个性指定带领B编号):
            return str(个性指定带领B编号).strip()

    return 渠道B编号


# ============================================================
# 六.5、特殊渠道带领指定（最高优先级，仅覆盖渠道B / 带领B）
# ============================================================
# 字典格式：
# {
#     "报名学员编号": ["渠道B编号", "带领B编号"]
# }
# 命中规则（最简）：
#   只要 key 存在，就视为命中，直接采用 value[0] / value[1]
#   即使 value[0] / value[1] 是空串，也照样使用，不再做合法编号过滤
# 防御性兜底（仅为避免脏数据导致代码崩溃，不算业务校验）：
#   - value 不是 list/tuple，或长度 < 2 -> 视为未命中
# 返回约定：
#   - 命中  : 返回 (str, str)，可能为空串
#   - 未命中: 返回 (None, None)，调用方据此决定回退到老逻辑

def 获取特殊指定渠道B带领B(报名学员编号):
    """
    根据报名学员编号，从特殊渠道带领指定映射中取出 (渠道B编号, 带领B编号)。
    命中条件：仅看 key 是否存在于映射字典中；value 内容不做任何业务过滤。
    """

    键 = "" if 报名学员编号 is None else str(报名学员编号).strip()

    if 键 not in 特殊渠道带领指定学员编号映射_渠道B_带领B:
        return None, None

    指定值 = 特殊渠道带领指定学员编号映射_渠道B_带领B.get(键)

    # 仅做最低限度的脏数据防御，不算业务校验
    if not isinstance(指定值, (list, tuple)) or len(指定值) < 2:
        return None, None

    指定渠道B原值 = 指定值[0]
    指定带领B原值 = 指定值[1]

    指定渠道B = "" if 指定渠道B原值 is None else str(指定渠道B原值).strip()
    指定带领B = "" if 指定带领B原值 is None else str(指定带领B原值).strip()

    return 指定渠道B, 指定带领B

def 查找上层合伙宝妈编号(起始报名学员编号):
    """
    以一个编号作为【报名学员编号】继续向上查找，直到找到有效的上层合伙宝妈编号。

    规则：
    1. 先用当前报名学员编号去查它的推荐人编号
    2. 如果这个推荐人编号不是合伙宝妈，则继续向上找
    3. 如果这个推荐人编号是合伙宝妈：
       - 若【当前节点编号】不在这个合伙宝妈的前五学员中，则返回这个合伙宝妈编号
       - 若【当前节点编号】在这个合伙宝妈的前五学员中，则继续向上找这个合伙宝妈的推荐人编号
    """

    if not 是合法编号(起始报名学员编号):
        return ""

    原始报名学员编号 = str(起始报名学员编号).strip()
    当前报名学员编号 = 原始报名学员编号

    已访问编号集合 = set()

    while True:
        if 当前报名学员编号 in 已访问编号集合:
            return ""

        已访问编号集合.add(当前报名学员编号)

        if not 是合法编号(当前报名学员编号):
            return ""

        当前报名学员编号 = str(当前报名学员编号).strip()

        上一级推荐人编号 = 学员来源映射.get(当前报名学员编号, "")

        if not 是合法编号(上一级推荐人编号):
            return ""

        上一级推荐人编号 = str(上一级推荐人编号).strip()

        前五学员编号列表 = 宝妈前五学员映射.get(上一级推荐人编号)

        # 当前推荐人不是合伙宝妈，继续向上找
        if 前五学员编号列表 is None:
            当前报名学员编号 = 上一级推荐人编号
            continue

        合法前五学员编号集合 = set()

        for 前五学员编号 in 前五学员编号列表:
            if 是合法编号(前五学员编号):
                合法前五学员编号集合.add(str(前五学员编号).strip())

        # 当前节点编号不在前五，则这个合伙宝妈就是结果
        if 当前报名学员编号 not in 合法前五学员编号集合:
            return 上一级推荐人编号

        # 当前节点编号在前五，继续向上找
        当前报名学员编号 = 上一级推荐人编号


# ------------------------------------------------------------
# 七、续：基于"再走一遍渠道B链路"的渠道A / 带领A 计算
# ------------------------------------------------------------
# 不再使用 查找上层合伙宝妈编号(...)；改为：
#   渠道A = 把 渠道B编号 当成新报名学员编号，再调一次 查找推荐人和渠道B(...)，取其渠道B
#   带领A = 把 带领B编号 当成新报名学员编号，再调一次 查找推荐人和渠道B(...)；
#         若命中特殊指定带领B 则直接用之，否则按 计算带领B编号(带领B编号, 新渠道B编号) 计算
# 注：查找上层合伙宝妈编号(...) 暂保留不删，避免影响其它潜在调用方。

def 计算渠道A编号(渠道B编号):
    """
    以 渠道B编号 为新起点，循环向上调用 查找推荐人和渠道B(...)，
    直到拿到一个 != 原始渠道B编号 的候选结果作为渠道A。
    若始终找不到（链路尽头 / 候选不合法 / 防环命中），返回空串。
    每轮若候选 == 原B，则把起点沿 学员来源映射 推进到上一级推荐人后继续。
    """
    if not 是合法编号(渠道B编号):
        return ""

    原始渠道B编号 = str(渠道B编号).strip()
    当前起点编号 = 原始渠道B编号
    已访问起点集合 = set()

    while True:
        if not 是合法编号(当前起点编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算渠道A-返回空] 原始渠道B={}，当前起点不合法，当前起点={}".format(
                        原始渠道B编号, 当前起点编号
                    )
                )
            return ""

        if 当前起点编号 in 已访问起点集合:
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算渠道A-返回空] 原始渠道B={}，当前起点={} 已访问过（防环）".format(
                        原始渠道B编号, 当前起点编号
                    )
                )
            return ""

        已访问起点集合.add(当前起点编号)

        _推荐人_忽略, 新渠道B编号, _指定带领B_忽略 = 查找推荐人和渠道B(当前起点编号)

        if 调试日志_查找推荐人和渠道B:
            print(
                "[计算渠道A] 原始渠道B={}，当前起点={}，候选渠道A={}".format(
                    原始渠道B编号, 当前起点编号, 新渠道B编号
                )
            )

        if not 是合法编号(新渠道B编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算渠道A-返回空] 原始渠道B={}，候选不合法/为空，结束查找".format(
                        原始渠道B编号
                    )
                )
            return ""

        新渠道B编号 = str(新渠道B编号).strip()

        if 新渠道B编号 != 原始渠道B编号:
            return 新渠道B编号

        # 候选与原B 相同 → 把起点沿"推荐人"链路向上推进，继续找
        上一级推荐人编号 = 学员来源映射.get(当前起点编号, "")
        if 调试日志_查找推荐人和渠道B:
            print(
                "[计算渠道A-继续向上] 原始渠道B={}，候选与原B相同，起点上移到={}".format(
                    原始渠道B编号, 上一级推荐人编号
                )
            )
        if not 是合法编号(上一级推荐人编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算渠道A-返回空] 原始渠道B={}，无合法上一级推荐人，结束查找".format(
                        原始渠道B编号
                    )
                )
            return ""

        当前起点编号 = str(上一级推荐人编号).strip()


def 计算带领A编号(带领B编号):
    """
    以 带领B编号 为新起点，循环：
      - 调用 查找推荐人和渠道B(当前起点)，命中"指定带领B" 则候选 = 指定带领B；
        否则候选 = 计算带领B编号(当前起点, 新渠道B编号)。
      - 候选不合法 -> 返回空串
      - 候选 != 原始带领B编号 -> 返回候选作为带领A
      - 候选 == 原始带领B编号 -> 起点沿 学员来源映射 推进到上一级推荐人后继续找
    防环：已访问的起点不再重复处理。
    """
    if not 是合法编号(带领B编号):
        return ""

    原始带领B编号 = str(带领B编号).strip()
    当前起点编号 = 原始带领B编号
    已访问起点集合 = set()

    while True:
        if not 是合法编号(当前起点编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算带领A-返回空] 原始带领B={}，当前起点不合法，当前起点={}".format(
                        原始带领B编号, 当前起点编号
                    )
                )
            return ""

        if 当前起点编号 in 已访问起点集合:
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算带领A-返回空] 原始带领B={}，当前起点={} 已访问过（防环）".format(
                        原始带领B编号, 当前起点编号
                    )
                )
            return ""

        已访问起点集合.add(当前起点编号)

        _推荐人_忽略, 新渠道B编号, 新指定带领B编号 = 查找推荐人和渠道B(当前起点编号)

        if 新指定带领B编号 is not None:
            候选带领A = 新指定带领B编号
        else:
            候选带领A = 计算带领B编号(当前起点编号, 新渠道B编号)

        if 调试日志_查找推荐人和渠道B:
            print(
                "[计算带领A] 原始带领B={}，当前起点={}，候选带领A={}".format(
                    原始带领B编号, 当前起点编号, 候选带领A
                )
            )

        if not 是合法编号(候选带领A):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算带领A-返回空] 原始带领B={}，候选不合法/为空，结束查找".format(
                        原始带领B编号
                    )
                )
            return ""

        候选带领A = str(候选带领A).strip()

        if 候选带领A != 原始带领B编号:
            return 候选带领A

        # 候选与原带B 相同 → 把起点沿"推荐人"链路向上推进，继续找
        上一级推荐人编号 = 学员来源映射.get(当前起点编号, "")
        if 调试日志_查找推荐人和渠道B:
            print(
                "[计算带领A-继续向上] 原始带领B={}，候选与原带B相同，起点上移到={}".format(
                    原始带领B编号, 上一级推荐人编号
                )
            )
        if not 是合法编号(上一级推荐人编号):
            if 调试日志_查找推荐人和渠道B:
                print(
                    "[计算带领A-返回空] 原始带领B={}，无合法上一级推荐人，结束查找".format(
                        原始带领B编号
                    )
                )
            return ""

        当前起点编号 = str(上一级推荐人编号).strip()


# ============================================================
# 七.5、新增学员触发"动态晋升合伙宝妈"
# ============================================================
# 调用时机：主流程遍历到一个"新增学员编号"时，调用本函数对其推荐人做一次预处理。
# 可能更新的运行态变量：
#   - 推荐人编号映射推荐的学员列表（清洗 + 必要时追加当前新增学员）
#   - 合伙宝妈编号列表（满足条件时追加新晋升的合伙宝妈）
#   - 宝妈前五学员映射（满足条件时写入新合伙宝妈 -> 前 5 学员）
# 不会修改 飞书宝妈前五学员映射（快照保持不变）。

def 尝试动态晋升合伙宝妈(报名学员编号):
    """
    根据需求文档的 B1 / B2 / B3 三种分支：
      - 推荐人不合法 / 已经是合伙宝妈 -> 直接返回，由主流程继续走原有流程
      - 否则取 推荐人编号映射推荐的学员列表[推荐人] 的 value（兜底为 []，仅保留合法 6 位编号且去重）
      - 长度 < 5：先追加当前报名学员（若未存在），追加后若达到 5 则晋升
      - 长度 == 5：直接晋升（取前 5）
      - 长度  > 5：直接晋升（取前 5）
    """

    if not 是合法编号(报名学员编号):
        return

    报名学员编号 = str(报名学员编号).strip()

    推荐人编号 = 学员来源映射.get(报名学员编号, "")
    if not 是合法编号(推荐人编号):
        return
    推荐人编号 = str(推荐人编号).strip()

    # 已经是合伙宝妈 -> 直接走原流程
    if 推荐人编号 in 宝妈前五学员映射:
        return

    原始推荐学员列表 = 推荐人编号映射推荐的学员列表.get(推荐人编号, [])
    if not isinstance(原始推荐学员列表, list):
        原始推荐学员列表 = []

    # 仅保留合法 6 位编号，统一 strip，并去重
    清洗后的推荐学员列表 = []
    已存在的推荐学员集合 = set()
    for _编号 in 原始推荐学员列表:
        if not 是合法编号(_编号):
            continue
        _编号 = str(_编号).strip()
        if _编号 in 已存在的推荐学员集合:
            continue
        清洗后的推荐学员列表.append(_编号)
        已存在的推荐学员集合.add(_编号)

    # 长度 < 5：先追加当前新增学员（若未存在）
    # 长度 >= 5：不追加（已经满 / 已超）
    if len(清洗后的推荐学员列表) < 5:
        if 报名学员编号 not in 已存在的推荐学员集合:
            清洗后的推荐学员列表.append(报名学员编号)
            已存在的推荐学员集合.add(报名学员编号)

    # 回写运行态：推荐人 -> 推荐学员列表（清洗后的版本）
    推荐人编号映射推荐的学员列表[推荐人编号] = 清洗后的推荐学员列表

    # 满足晋升条件：长度 >= 5（含正好 == 5 与 > 5 两种边界）
    if len(清洗后的推荐学员列表) >= 5:
        前5编号 = 清洗后的推荐学员列表[:5]
        宝妈前五学员映射[推荐人编号] = 前5编号
        if 推荐人编号 not in 合伙宝妈编号列表:
            合伙宝妈编号列表.append(推荐人编号)
        本轮新增合伙宝妈编号集合.add(推荐人编号)

        print(
            "[动态晋升-新增触发] 推荐人编号={}，原推荐学员数={}，飞书未同步，补录为合伙宝妈".format(
                推荐人编号, len(清洗后的推荐学员列表)
            )
        )
        print(
            "[动态晋升-新增触发] 已写入宝妈前五学员映射：{} -> {}".format(推荐人编号, 前5编号)
        )


# ============================================================
# 八、主流程：遍历所有报名学员编号，写入 txt，并产出全局字典
# ============================================================
# txt 输出格式：
# 推荐人编号\t渠道B编号\t带领B编号\t渠道A编号\t带领A编号
#
# 全局字典格式：
# {
#     "报名学员编号": ["推荐人编号", "渠道B编号", "带领B编号", "渠道A编号", "带领A编号"]
# }

学员编号映射编号列表_推荐人_渠道B_带领B = {}

总数量 = 0
成功数量 = 0
空结果数量 = 0

with open(输出文件路径, "w", encoding="utf-8") as 文件:
    for 报名学员编号 in 数据库学员编号列表:
        总数量 += 1

        推荐人编号 = ""
        渠道B编号 = ""
        带领B编号 = ""
        渠道A编号 = ""
        带领A编号 = ""

        原始报名学员编号 = "" if 报名学员编号 is None else str(报名学员编号).strip()

        if 是合法编号(报名学员编号):
            # 新增学员触发"动态晋升合伙宝妈"预处理：
            # 仅当本次报名学员属于 新增学员（数据库有、飞书无）时，
            # 才尝试根据 推荐人编号映射推荐的学员列表 动态更新
            # 合伙宝妈编号列表 / 宝妈前五学员映射，
            # 让随后调用的"原有流程"在同一轮里就能吃到最新状态。
            if 原始报名学员编号 in 新增学员编号集合:
                尝试动态晋升合伙宝妈(原始报名学员编号)

            # 推荐人编号始终走现有逻辑；渠道B / 带领B 是否走特殊映射，
            # 由 查找推荐人和渠道B(...) 在向上查找过程中判断（链路上任一节点命中即返回）
            推荐人编号, 渠道B编号, 指定带领B编号 = 查找推荐人和渠道B(原始报名学员编号)

            if 指定带领B编号 is not None:
                # 渠道B 查找过程中命中了特殊映射；渠道B编号 已经是特殊指定值
                # 带领B 直接采用特殊指定值，不再调用原 计算带领B编号(...)
                带领B编号 = 指定带领B编号
            else:
                # 未命中特殊映射，带领B 走原计算逻辑
                带领B编号 = 计算带领B编号(原始报名学员编号, 渠道B编号)

            # 再计算渠道A编号、带领A编号（算法不变，输入用最终的渠道B / 带领B）
            # 若最终渠道B / 带领B 为空串，下面会自然返回空串，无需新增规则
            渠道A编号 = 计算渠道A编号(渠道B编号)
            带领A编号 = 计算带领A编号(带领B编号)

            # 排查日志：推荐人有值但渠道B 为空（最常见的异常组合）
            if 推荐人编号 and not 渠道B编号:
                print(
                    "[结果异常排查] 报名学员={}，推荐人={}，渠道B为空，带领B为空，渠道A为空，带领A为空".format(
                        原始报名学员编号, 推荐人编号
                    )
                )

        # 无论编号是否合法，都写入全局字典
        学员编号映射编号列表_推荐人_渠道B_带领B[原始报名学员编号] = [
            推荐人编号,
            渠道B编号,
            带领B编号,
            渠道A编号,
            带领A编号,
        ]

        # 保留写入 txt 文件逻辑
        文件.write(
            推荐人编号 + "\t" +
            渠道B编号 + "\t" +
            带领B编号 + "\t" +
            渠道A编号 + "\t" +
            带领A编号 + "\n"
        )
